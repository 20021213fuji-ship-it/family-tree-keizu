        // ユリウス通日(JDN)計算（グレゴリオ暦 / 1582-10-15以前はユリウス暦）
        function toJDN(year, month, day) {
            const a = Math.floor((14 - month) / 12);
            const y = year + 4800 - a;
            const m = month + 12 * a - 3;
            const isGregorian = (year > 1582) || (year === 1582 && (month > 10 || (month === 10 && day >= 15)));
            if (isGregorian) {
                return day + Math.floor((153 * m + 2) / 5) + 365 * y + Math.floor(y / 4) - Math.floor(y / 100) + Math.floor(y / 400) - 32045;
            } else {
                return day + Math.floor((153 * m + 2) / 5) + 365 * y + Math.floor(y / 4) - 32083;
            }
        }

        function fromJDN(jdn) {
            let a = jdn + 32044;
            let b = Math.floor((4 * a + 3) / 146097);
            let c = a - Math.floor((146097 * b) / 4);
            let d = Math.floor((4 * c + 3) / 1461);
            let e = c - Math.floor((1461 * d) / 4);
            let m = Math.floor((5 * e + 2) / 153);
            const day = e - Math.floor((153 * m + 2) / 5) + 1;
            const month = m + 3 - 12 * Math.floor(m / 10);
            const year = 100 * b + d - 4800 + Math.floor(m / 10);
            return { year, month, day };
        }

        // 1700年より前の旧暦近似計算用（天文平均朔望周期による換算）
        function approxGregorianToLunar(year, month, day) {
            const jdn = toJDN(year, month, day);
            const baseJdn = toJDN(1700, 2, 19); // 1700年2月19日 = 元禄13年1月1日
            const synodicMonth = 29.530588853;
            const diffDays = jdn - baseJdn;
            const totalMonths = Math.floor(diffDays / synodicMonth);
            const dayInMonth = Math.max(1, Math.min(30, Math.floor(diffDays - totalMonths * synodicMonth) + 1));
            const approxYear = 1700 + Math.floor(totalMonths / 12.368266);
            let approxMonth = (((totalMonths % 12) + 12) % 12) + 1;
            if (month <= 2 && approxMonth >= 11) {
                return { lunarYear: year - 1, lunarMonth: approxMonth, lunarDay: dayInMonth, isLeap: false };
            }
            return { lunarYear: approxYear, lunarMonth: approxMonth, lunarDay: dayInMonth, isLeap: false };
        }

        // 西暦（1872年12月31日まで）→ 旧暦（年・月・日・閏月フラグ）に変換（国立天文台暦日テーブル準拠）
        function gregorianToLunar(year, month, day) {
            const gNum = year * 10000 + (month || 1) * 100 + (day || 1);
            // 1873年1月1日（新暦明治6年1月1日）以降は旧暦対象外
            if (gNum >= 18730101) return null;

            if (!month || !day) {
                const ly = (month && month <= 2) ? year - 1 : year;
                return { lunarYear: ly, lunarMonth: month || null, lunarDay: day || null, isLeap: false };
            }

            const targetJdn = toJDN(year, month, day);
            if (year >= 1700 && year <= 1872) {
                let lunarYear = year;
                const entryThis = NAO_LUNAR_CALENDAR[year];
                if (entryThis) {
                    const nyJdn = toJDN(year, entryThis[0], entryThis[1]);
                    if (targetJdn < nyJdn) lunarYear = year - 1;
                }
                const entry = NAO_LUNAR_CALENDAR[lunarYear];
                if (entry) {
                    const [nyMonth, nyDay, leapMonth, daysArr] = entry;
                    let remDays = targetJdn - toJDN(lunarYear, nyMonth, nyDay);
                    if (remDays >= 0) {
                        for (let idx = 0; idx < daysArr.length; idx++) {
                            const mLen = daysArr[idx];
                            if (remDays < mLen) {
                                let lunarMonth;
                                let isLeap = false;
                                if (leapMonth > 0) {
                                    if (idx < leapMonth) {
                                        lunarMonth = idx + 1;
                                    } else if (idx === leapMonth) {
                                        lunarMonth = leapMonth;
                                        isLeap = true;
                                    } else {
                                        lunarMonth = idx;
                                    }
                                } else {
                                    lunarMonth = idx + 1;
                                }
                                return {
                                    lunarYear,
                                    lunarMonth,
                                    lunarDay: remDays + 1,
                                    isLeap
                                };
                            }
                            remDays -= mLen;
                        }
                    }
                }
            }

            return approxGregorianToLunar(year, month, day);
        }

        // 旧暦（年・月・日・閏月）→ グレゴリオ暦（西暦）に変換（国立天文台暦日テーブル準拠）
        function lunarToGregorian(lunarYear, lunarMonth, lunarDay, isLeap = false) {
            if (!lunarMonth || !lunarDay) {
                return { year: lunarYear, month: lunarMonth || null, day: lunarDay || null };
            }
            if (lunarYear >= 1700 && lunarYear <= 1872) {
                const entry = NAO_LUNAR_CALENDAR[lunarYear];
                if (entry) {
                    const [nyMonth, nyDay, leapMonth, daysArr] = entry;
                    let offsetDays = 0;
                    for (let idx = 0; idx < daysArr.length; idx++) {
                        let curM, curLeap = false;
                        if (leapMonth > 0) {
                            if (idx < leapMonth) {
                                curM = idx + 1;
                            } else if (idx === leapMonth) {
                                curM = leapMonth;
                                curLeap = true;
                            } else {
                                curM = idx;
                            }
                        } else {
                            curM = idx + 1;
                        }
                        if (curM === lunarMonth && curLeap === isLeap) {
                            offsetDays += Math.max(0, lunarDay - 1);
                            const gJdn = toJDN(lunarYear, nyMonth, nyDay) + offsetDays;
                            return fromJDN(gJdn);
                        }
                        offsetDays += daysArr[idx];
                    }
                }
            }
            // 1700年以前は概算で返す
            const baseJdn = toJDN(1700, 2, 19);
            const approxDays = Math.round((lunarYear - 1700) * 365.2422 + (lunarMonth - 1) * 29.5306 + (lunarDay - 1));
            return fromJDN(baseJdn + approxDays);
        }

        // 旧暦用の元号検索（明治5年12月2日まで）
        function findLunarEra(lunarYear, lunarMonth, lunarDay) {
            const m = lunarMonth || 12;
            const d = lunarDay || 30;
            const num = lunarYear * 10000 + m * 100 + d;

            for (const era of LUNAR_ERAS) {
                const startNum = era.startYear * 10000 + era.startMonth * 100 + era.startDay;
                if (num >= startNum) {
                    return { name: era.name, year: lunarYear - era.startYear + 1 };
                }
            }
            return null;
        }

        // 新暦用の元号検索（明治6年1月1日以降）
        function findJapaneseEra(year, month, day) {
            const m = month || 12;
            const d = day || 31;
            const num = year * 10000 + m * 100 + d;

            for (const era of ALL_ERAS) {
                const startNum = era.startYear * 10000 + era.startMonth * 100 + era.startDay;
                if (num >= startNum) {
                    return { name: era.name, year: year - era.startYear + 1 };
                }
            }
            return null;
        }

        // 文字列を解析して { year, month, day, isWarekiInput, isLeap, lunarInfo, raw } を返す
        function parseDateInput(str) {
            if (!str || !str.trim()) return null;
            const s = str.trim()
                .replace(/[０-９]/g, ch => String.fromCharCode(ch.charCodeAt(0) - 0xFEE0))
                .replace(/元年/g, '1年');

            // 1. 和暦（旧暦または新暦の和暦）入力パターン
            const candidateEras = [...LUNAR_ERAS, ...ALL_ERAS, ...EXTRA_PARSE_ERAS]
                .filter((v, i, a) => a.findIndex(x => x.name === v.name) === i)
                .sort((a, b) => b.name.length - a.name.length);

            for (const era of candidateEras) {
                if (s.startsWith(era.name)) {
                    const rest = s.slice(era.name.length);
                    const m = rest.match(/^(\d{1,3})(?:[年\.\/\-](閏?)(\d{1,2}))?(?:[月\.\/\-](\d{1,2}))?/);
                    if (m) {
                        const eraYear = parseInt(m[1], 10);
                        const baseYear = era.startYear + eraYear - 1;
                        const isLeap = m[2] === '閏';
                        const month = m[3] ? parseInt(m[3], 10) : null;
                        const day = m[4] ? parseInt(m[4], 10) : null;

                        // 明治5年12月2日以前なら旧暦として扱う
                        const checkNum = baseYear * 10000 + (month || 1) * 100 + (day || 1);
                        const isOldLunar = checkNum <= 18721202;

                        if (isOldLunar) {
                            const greg = lunarToGregorian(baseYear, month, day, isLeap);
                            return {
                                year: greg.year,
                                month: greg.month,
                                day: greg.day,
                                isWarekiInput: true,
                                isOldLunar: true,
                                lunarInfo: {
                                    eraName: era.name,
                                    eraYear,
                                    lunarYear: baseYear,
                                    lunarMonth: month,
                                    lunarDay: day,
                                    isLeap
                                },
                                raw: str.trim()
                            };
                        } else {
                            return {
                                year: baseYear,
                                month,
                                day,
                                isWarekiInput: true,
                                isOldLunar: false,
                                lunarInfo: null,
                                raw: str.trim()
                            };
                        }
                    }
                }
            }

            // 2. 西暦パターン (例: 1852-3-5, 1940-05-01, 1872/12/31)
            const adMatch = s.match(/^(\d{3,4})(?:[年\.\/\-](\d{1,2}))?(?:[月\.\/\-](\d{1,2}))?/);
            if (adMatch) {
                const year = parseInt(adMatch[1], 10);
                const month = adMatch[2] ? parseInt(adMatch[2], 10) : null;
                const day = adMatch[3] ? parseInt(adMatch[3], 10) : null;
                const checkNum = year * 10000 + (month || 1) * 100 + (day || 1);
                const isOldLunar = checkNum <= 18721231;

                return {
                    year,
                    month,
                    day,
                    isWarekiInput: false,
                    isOldLunar,
                    lunarInfo: isOldLunar ? gregorianToLunar(year, month, day) : null,
                    raw: str.trim()
                };
            }

            return { year: null, month: null, day: null, isWarekiInput: false, isOldLunar: false, lunarInfo: null, raw: str.trim() };
        }

        // 設定（西暦/和暦）に合わせて日付を表示文字列にフォーマット
        // ※1872年12月31日（旧暦 明治5年12月2日）までは旧暦で表示し、翌1873年1月1日からは新暦表示
        function formatDisplayDate(str, mode = appState.calendarMode) {
            const parsed = parseDateInput(str);
            if (!parsed) return '';
            if (!parsed.year) return parsed.raw;

            // 1. 「西暦」モードのときは従来通り西暦で表示
            if (mode === 'ad') {
                if (parsed.month && parsed.day) {
                    return `${parsed.year}/${String(parsed.month).padStart(2, '0')}/${String(parsed.day).padStart(2, '0')}`;
                }
                if (parsed.month) return `${parsed.year}年${parsed.month}月`;
                return `${parsed.year}年`;
            }

            // 2. 「和暦」モードのとき
            // 2-A. 旧暦（明治5年12月2日＝西暦1872年12月31日まで）
            if (parsed.isOldLunar && parsed.lunarInfo) {
                const li = parsed.lunarInfo;
                // 旧暦（和暦）で直接入力された場合はその元号・年月日をそのまま表示
                if (parsed.isWarekiInput && li.eraName) {
                    const eraYearStr = li.eraYear === 1 ? '元年' : `${li.eraYear}年`;
                    const leapPrefix = li.isLeap ? '閏' : '';
                    if (li.lunarMonth && li.lunarDay) {
                        return `${li.eraName}${eraYearStr}${leapPrefix}${li.lunarMonth}月${li.lunarDay}日`;
                    }
                    if (li.lunarMonth) {
                        return `${li.eraName}${eraYearStr}${leapPrefix}${li.lunarMonth}月`;
                    }
                    return `${li.eraName}${eraYearStr}`;
                }

                // 西暦で入力された場合は旧暦の元号・旧暦年月日に変換して表示（例: 1852-3-5 -> 嘉永5年2月15日）
                const era = findLunarEra(li.lunarYear, li.lunarMonth, li.lunarDay);
                const leapPrefix = li.isLeap ? '閏' : '';
                if (!era) {
                    if (li.lunarMonth && li.lunarDay) return `旧暦${li.lunarYear}年${leapPrefix}${li.lunarMonth}月${li.lunarDay}日`;
                    return `旧暦${li.lunarYear}年`;
                }
                const eraYearStr = era.year === 1 ? '元年' : `${era.year}年`;
                if (li.lunarMonth && li.lunarDay) {
                    return `${era.name}${eraYearStr}${leapPrefix}${li.lunarMonth}月${li.lunarDay}日`;
                }
                if (li.lunarMonth) {
                    return `${era.name}${eraYearStr}${leapPrefix}${li.lunarMonth}月`;
                }
                return `${era.name}${eraYearStr}`;
            }

            // 2-B. 新暦（1873年1月1日＝明治6年1月1日以降）
            const era = findJapaneseEra(parsed.year, parsed.month, parsed.day);
            if (!era) {
                return parsed.month && parsed.day ? `${parsed.year}年${parsed.month}月${parsed.day}日` : `${parsed.year}年`;

        function getActiveLineage() {

                    appState.activeLineageId = keys[0];

        // 名字と名前から表示用フルネームを組み立てる
        function getPersonDisplayName(p) {
            if (!p) return '';
            const ln = (p.lastName || '').trim();
            const fn = (p.firstName || '').trim();
            if (ln && fn) return `${ln} ${fn}`;
            if (ln || fn) return ln || fn;
            return p.name || '';
        }

        // 既存データのマイグレーション（名字・名前分割、adoptiveParentIds初期化）
        function normalizePersonData(p) {
            if (p.lastName === undefined && p.firstName === undefined) {
                const full = (p.name || '').trim();
                const parts = full.split(/[\s　]+/);
                if (parts.length >= 2) {
                    p.lastName = parts[0];
                    p.firstName = parts.slice(1).join(' ');
                } else {
                    p.lastName = full;
                    p.firstName = '';
                }
            }
            if (p.dharmaName === undefined) p.dharmaName = '';
            if (p.lifeStatus === undefined) {
                p.lifeStatus = (p.deathDate && p.deathDate.trim() !== '') ? 'deceased' : 'alive';
            }
            if (!Array.isArray(p.parentIds)) p.parentIds = [];
            if (!Array.isArray(p.adoptiveParentIds)) p.adoptiveParentIds = [];
            if (!Array.isArray(p.spouseIds)) p.spouseIds = [];
            if (!Array.isArray(p.childrenIds)) p.childrenIds = [];
            p.name = getPersonDisplayName(p);
            return p;
        }

        // 指定した親IDが「養親」かどうかを判定
        function isAdoptiveParent(childPerson, parentId) {
            if (!childPerson || !Array.isArray(childPerson.adoptiveParentIds)) return false;
            return childPerson.adoptiveParentIds.includes(parentId);
        }

        // 人物が故人かどうかを判定
        function isPersonDeceased(p) {
            if (!p) return false;
            if (p.deathDate && p.deathDate.trim() !== '') return true;
            return p.lifeStatus === 'deceased';
        }

        function switchBottomTab(tabKey) {
            appState.activeBottomTab = tabKey;

            const views = {
                tree: document.getElementById('tab-view-tree'),
                history: document.getElementById('tab-view-history'),
                people: document.getElementById('tab-view-people'),
                docs: document.getElementById('tab-view-docs'),

            const buttons = {
                tree: document.getElementById('bottom-tab-tree'),
                history: document.getElementById('bottom-tab-history'),
                people: document.getElementById('bottom-tab-people'),
                docs: document.getElementById('bottom-tab-docs'),
                map: document.getElementById('bottom-tab-map')
            };

            const activeBtnClass = 'flex-1 flex flex-col items-center justify-center py-1 rounded-xl text-indigo-600 dark:text-indigo-400 bg-indigo-50/80 dark:bg-indigo-950/50 font-bold transition-colors cursor-pointer';

            Object.keys(views).forEach(key => {
                if (views[key]) {
                    if (key === tabKey) {
                        views[key].classList.remove('hidden');
                        views[key].classList.add('flex');
                    } else {
                        views[key].classList.add('hidden');
                        views[key].classList.remove('flex');
                    }
                }
                if (buttons[key]) {
                    buttons[key].className = key === tabKey ? activeBtnClass : inactiveBtnClass;
                }
            });

            if (tabKey === 'tree') {
                renderTree();
            } else if (tabKey === 'people') {
                renderPeopleView();
            } else if (tabKey === 'docs') {
                renderDocsView();
            } else if (tabKey === 'map') {
                renderMapView();
            }
            if (window.lucide) lucide.createIcons();
        }

        // 6. レイアウト計算エンジン（TB:上から下 / LR:左から右 & 横書/縦書）
        // ==========================================
        function getCardDimensions(p, isVertical) {
            const displayName = getPersonDisplayName(p);
            const charCount = Array.from(displayName).length;
            const birthDisp = formatDisplayDate(p.birthDate);
            const deathDisp = formatDisplayDate(p.deathDate);
            const isDeceased = isPersonDeceased(p);
            const noteClean = (p.note || '').trim();

            if (!isVertical) {
                const minNameW = Math.max(210, 74 + charCount * 17);
                const w = Math.min(340, minNameW);
                let h = 92;
                if (p.dharmaName) h += 20;
                if (birthDisp || deathDisp || isDeceased) h += 18;
                if (p.domicile) h += 18;
                if ((p.branchType && p.branchType !== 'none') || (p.inheritanceStatus && p.inheritanceStatus !== 'none') || p.branchDetail) h += 24;
                if (p.linkedLineageId && appState.lineages[p.linkedLineageId]) h += 26;
                if (noteClean) {
                    const lines = Math.ceil(noteClean.length / Math.max(12, Math.floor((w - 28) / 11)));
                    h += 14 + lines * 15;
                }
                return { w, h };
            } else {
                const longestToken = Math.max(
                    charCount,
                    Array.from(p.relationship || '').length,
                    Array.from(p.dharmaName || '').length,
                    8
                );
                const h = Math.min(260, Math.max(125, 42 + longestToken * 16));

                let w = 78;
                if (p.dharmaName) w += 20;
                if (birthDisp || deathDisp || isDeceased) w += 18;
                if (p.domicile) {
                    const domCols = Math.max(1, Math.ceil(Array.from(`本籍 ${p.domicile}`).length / Math.max(8, Math.floor((h - 24) / 12))));
                    w += domCols * 16;
                }
                if ((p.branchType && p.branchType !== 'none') || (p.inheritanceStatus && p.inheritanceStatus !== 'none') || p.branchDetail) {
                    w += 22;
                }
                if (noteClean) {
                    const charsPerCol = Math.max(8, Math.floor((h - 24) / 12));
                    const noteCols = Math.max(1, Math.ceil(Array.from(noteClean).length / charsPerCol));
                    w += 10 + noteCols * 15;
                }
                if (p.linkedLineageId && appState.lineages[p.linkedLineageId]) {
                    w += 24;
                }
                return { w: Math.max(96, w), h };
            }
        }

        function calculateLayout() {
            const people = getActivePeople();
            const ids = Object.keys(people);
            if (ids.length === 0) return { bounds: {}, width: 800, height: 600 };

            const isLR = appState.layoutDirection === 'LR';
            const isVertical = appState.writingMode === 'vertical';

            const SPOUSE_GAP = isLR ? 22 : 34;
            const UNIT_GAP = isLR ? 34 : 44;
            const LEVEL_GAP = isLR ? 85 : 80;
            const PADDING = 80;

            // 1. 世代レベルの算出
            const gen = {};
            ids.forEach(id => { gen[id] = 0; });

            for (let iter = 0; iter < ids.length + 3; iter++) {
                let changed = false;
                ids.forEach(id => {
                    const p = people[id];
                    (p.childrenIds || []).forEach(cid => {
                        if (people[cid] && gen[cid] <= gen[id]) {
                            gen[cid] = gen[id] + 1;
                            changed = true;
                        }
                    });
                    (p.spouseIds || []).forEach(sid => {
                        if (people[sid] && gen[sid] !== gen[id]) {
                            const maxG = Math.max(gen[id], gen[sid]);
                            gen[id] = maxG;
                            gen[sid] = maxG;
                            changed = true;
                        }
                    });
                });
                if (!changed) break;
            }

            // 2. 世代ごとにグループ化
            const levels = {};
            let maxGen = 0;
            ids.forEach(id => {
                const g = gen[id];
                if (g > maxGen) maxGen = g;
                if (!levels[g]) levels[g] = [];
                levels[g].push(id);
            });

            // 3. 各世代内で「夫婦ユニット」を形成
            const levelUnits = {};
            for (let g = 0; g <= maxGen; g++) {
                const list = levels[g] || [];
                const visited = new Set();
                const units = [];
                list.forEach(id => {
                    if (visited.has(id)) return;
                    const p = people[id];
                    const spouseInLevel = (p.spouseIds || []).find(sid => list.includes(sid) && !visited.has(sid));
                    if (spouseInLevel) {
                        visited.add(id);
                        visited.add(spouseInLevel);
                        units.push([id, spouseInLevel]);
                    } else {
                        visited.add(id);
                        units.push([id]);
                    }
                });
                levelUnits[g] = units;
            }

            // 4. 各カードの推定高さと幅を事前計算
            const cardDims = {};
            ids.forEach(id => {
                cardDims[id] = getCardDimensions(people[id], isVertical);
            });

            // 5. 世代ごとの「階層方向サイズ（TBなら高さ、LRなら幅）」を算出
            const levelPrimarySize = {};
            const levelPrimaryPos = {};
            let currentPrimary = PADDING;

            for (let g = 0; g <= maxGen; g++) {
                const list = levels[g] || [];
                let maxDim = isLR ? 140 : 120;
                list.forEach(id => {
                    const dim = isLR ? cardDims[id].w : cardDims[id].h;
                    if (dim > maxDim) maxDim = dim;
                });
                levelPrimarySize[g] = maxDim;
                levelPrimaryPos[g] = currentPrimary;
                currentPrimary += maxDim + LEVEL_GAP;
            }

            // 6. 各ユニット内の「並列方向サイズ（TBなら幅、LRなら高さ）」を計算
            const unitSecondarySizes = {};
            const levelTotalSecondary = {};
            let maxSecondaryExtent = 0;

            for (let g = 0; g <= maxGen; g++) {
                const units = levelUnits[g];
                let totalSec = 0;
                unitSecondarySizes[g] = units.map(unit => {
                    const sec = unit.reduce((acc, id) => acc + (isLR ? cardDims[id].h : cardDims[id].w), 0)
                              + (unit.length > 1 ? SPOUSE_GAP : 0);
                    return sec;
                });
                totalSec = unitSecondarySizes[g].reduce((a, b) => a + b, 0) + Math.max(0, units.length - 1) * UNIT_GAP;
                levelTotalSecondary[g] = totalSec;
                if (totalSec > maxSecondaryExtent) maxSecondaryExtent = totalSec;
            }

            // 7. 親の位置に基づくソートと座標確定
            const bounds = {};
            const tempSecondaryCenter = {};

            for (let g = 0; g <= maxGen; g++) {
                const units = levelUnits[g];
                if (g > 0) {
                    const combined = units.map((u, idx) => ({ unit: u, size: unitSecondarySizes[g][idx] }));
                    combined.sort((a, b) => {
                        const getParentCenter = (unit) => {
                            const centers = [];
                            unit.forEach(id => {
                                (people[id].parentIds || []).forEach(pid => {
                                    if (tempSecondaryCenter[pid] !== undefined) centers.push(tempSecondaryCenter[pid]);
                                });
                            });
                            if (centers.length === 0) return 999999;
                            return centers.reduce((acc, v) => acc + v, 0) / centers.length;
                        };
                        return getParentCenter(a.unit) - getParentCenter(b.unit);
                    });
                    levelUnits[g] = combined.map(c => c.unit);
                    unitSecondarySizes[g] = combined.map(c => c.size);
                }

                let curSec = PADDING + (maxSecondaryExtent - levelTotalSecondary[g]) / 2;
                levelUnits[g].forEach((unit) => {
                    unit.forEach(id => {
                        const w = cardDims[id].w;
                        const h = cardDims[id].h;
                        if (!isLR) {
                            const x = curSec;
                            const y = levelPrimaryPos[g];
                            bounds[id] = { x, y, w, h };
                            tempSecondaryCenter[id] = x + w / 2;
                            curSec += w + SPOUSE_GAP;
                        } else {
                            const x = levelPrimaryPos[g];
                            const y = curSec;
                            bounds[id] = { x, y, w, h };
                            tempSecondaryCenter[id] = y + h / 2;
                            curSec += h + SPOUSE_GAP;
                        }
                    });
                    curSec -= SPOUSE_GAP;
                    curSec += UNIT_GAP;
                });
            }

            const totalWidth = isLR ? (currentPrimary - LEVEL_GAP + PADDING) : (maxSecondaryExtent + PADDING * 2);
            const totalHeight = isLR ? (maxSecondaryExtent + PADDING * 2) : (currentPrimary - LEVEL_GAP + PADDING);

            return { bounds, width: Math.max(totalWidth, 800), height: Math.max(totalHeight, 600) };
        }

        // ==========================================
        // 7. 家系図の描画（カード＆接続線：実親子＝実線 / 養親子＝点線）
        // ==========================================
        const BRANCH_LABELS = {

        function renderTree() {
            const lineage = getActiveLineage();
            const people = lineage.people || {};
            const ids = Object.keys(people);

            ids.forEach(id => normalizePersonData(people[id]));

            const nodesContainer = document.getElementById('nodes-container');
            const svgLayer = document.getElementById('connections-svg');
            const emptyState = document.getElementById('empty-state');
            const badge = document.getElementById('current-lineage-badge');

            if (badge) badge.textContent = lineage.name;
            updateModeBadge();

            nodesContainer.innerHTML = '';
            svgLayer.innerHTML = '';

            if (ids.length === 0) {
                document.getElementById('empty-lineage-title').textContent = `「${lineage.name}」に人物がいません`;
                emptyState.classList.remove('hidden');
                cardBoundsMap = {};
                return;
            } else {
                emptyState.classList.add('hidden');
            }

            const isLR = appState.layoutDirection === 'LR';
            const isVertical = appState.writingMode === 'vertical';
            const { bounds, width, height } = calculateLayout();
            cardBoundsMap = bounds;

            svgLayer.setAttribute('width', width);
            svgLayer.setAttribute('height', height);
            nodesContainer.style.width = `${width}px`;
            nodesContainer.style.height = `${height}px`;

            // 1. 婚姻線の描画（二重線）
            const drawnMarriages = new Set();
            ids.forEach(id => {
                const p = people[id];
                const b1 = bounds[id];
                if (!b1) return;

                (p.spouseIds || []).forEach(sid => {
                    if (!people[sid] || !bounds[sid]) return;
                    const pairKey = [id, sid].sort().join('_');
                    if (drawnMarriages.has(pairKey)) return;
                    drawnMarriages.add(pairKey);

                    const b2 = bounds[sid];
                    if (!isLR) {
                        const left = b1.x < b2.x ? b1 : b2;
                        const right = b1.x < b2.x ? b2 : b1;
                        const x1 = left.x + left.w;
                        const x2 = right.x;
                        const y = Math.min(left.y + left.h / 2, right.y + right.h / 2);

                        const line1 = createSvgEl('line', { x1, y1: y - 2.5, x2, y2: y - 2.5, stroke: '#ec4899', 'stroke-width': '2' });
                        const line2 = createSvgEl('line', { x1, y1: y + 2.5, x2, y2: y + 2.5, stroke: '#ec4899', 'stroke-width': '2' });
                        svgLayer.appendChild(line1);
                        svgLayer.appendChild(line2);
                    } else {
                        const top = b1.y < b2.y ? b1 : b2;
                        const bottom = b1.y < b2.y ? b2 : b1;
                        const y1 = top.y + top.h;
                        const y2 = bottom.y;
                        const x = Math.min(top.x + top.w / 2, bottom.x + bottom.w / 2);

                        const line1 = createSvgEl('line', { x1: x - 2.5, y1, x2: x - 2.5, y2, stroke: '#ec4899', 'stroke-width': '2' });
                        const line2 = createSvgEl('line', { x1: x + 2.5, y1, x2: x + 2.5, y2, stroke: '#ec4899', 'stroke-width': '2' });
                        svgLayer.appendChild(line1);
                        svgLayer.appendChild(line2);
                    }
                });
            });

            // 2. 親子線の描画（実親グループ＝実線 / 養親グループ＝点線 に完全分離）
            const familyGroups = {};
            ids.forEach(cid => {
                const c = people[cid];
                const validParents = (c.parentIds || []).filter(pid => people[pid] && bounds[pid]);
                if (validParents.length > 0) {
                    const bioParents = validParents.filter(pid => !isAdoptiveParent(c, pid)).sort();
                    const adoptParents = validParents.filter(pid => isAdoptiveParent(c, pid)).sort();

                    if (bioParents.length > 0) {
                        const bioKey = `${bioParents.join('_')}__bio`;
                        if (!familyGroups[bioKey]) {
                            familyGroups[bioKey] = {
                                parents: bioParents,
                                children: [],
                                isAdoptive: false
                            };
                        }
                        familyGroups[bioKey].children.push(cid);
                    }

                    if (adoptParents.length > 0) {
                        const adoptKey = `${adoptParents.join('_')}__adopt`;
                        if (!familyGroups[adoptKey]) {
                            familyGroups[adoptKey] = {
                                parents: adoptParents,
                                children: [],
                                isAdoptive: true
                            };
                        }
                        familyGroups[adoptKey].children.push(cid);
                    }
                }
            });

            Object.values(familyGroups).forEach(group => {
                const pBounds = group.parents.map(pid => bounds[pid]);
                const cBounds = group.children.map(cid => bounds[cid]);
                const strokeColor = group.isAdoptive ? '#6366f1' : '#94a3b8';
                const dashAttrs = group.isAdoptive ? { 'stroke-dasharray': '5, 4' } : {};

                if (!isLR) {
                    let originX, originY;
                    if (pBounds.length >= 2) {
                        const left = pBounds[0].x < pBounds[1].x ? pBounds[0] : pBounds[1];
                        const right = pBounds[0].x < pBounds[1].x ? pBounds[1] : pBounds[0];
                        originX = (left.x + left.w + right.x) / 2;
                        originY = Math.min(left.y + left.h / 2, right.y + right.h / 2);
                    } else {
                        originX = pBounds[0].x + pBounds[0].w / 2;
                        originY = pBounds[0].y + pBounds[0].h;
                    }

                    const childTops = cBounds.map(cb => ({ x: cb.x + cb.w / 2, y: cb.y }));
                    const minChildY = Math.min(...childTops.map(pt => pt.y));
                    const midRatio = group.isAdoptive ? 0.42 : 0.52;
                    const midY = originY + (minChildY - originY) * midRatio;

                    svgLayer.appendChild(createSvgEl('line', {
                        x1: originX, y1: originY, x2: originX, y2: midY,
                        stroke: strokeColor, 'stroke-width': '2', 'stroke-linecap': 'round', ...dashAttrs
                    }));

                    const allX = [originX, ...childTops.map(pt => pt.x)];
                    svgLayer.appendChild(createSvgEl('line', {
                        x1: Math.min(...allX), y1: midY, x2: Math.max(...allX), y2: midY,
                        stroke: strokeColor, 'stroke-width': '2', 'stroke-linecap': 'round', ...dashAttrs
                    }));

                    childTops.forEach(pt => {
                        svgLayer.appendChild(createSvgEl('line', {
                            x1: pt.x, y1: midY, x2: pt.x, y2: pt.y,
                            stroke: strokeColor, 'stroke-width': '2', 'stroke-linecap': 'round', ...dashAttrs
                        }));
                    });
                } else {
                    let originX, originY;
                    if (pBounds.length >= 2) {
                        const top = pBounds[0].y < pBounds[1].y ? pBounds[0] : pBounds[1];
                        const bottom = pBounds[0].y < pBounds[1].y ? pBounds[1] : pBounds[0];
                        originX = Math.min(top.x + top.w / 2, bottom.x + bottom.w / 2);
                        originY = (top.y + top.h + bottom.y) / 2;
                    } else {
                        originX = pBounds[0].x + pBounds[0].w;
                        originY = pBounds[0].y + pBounds[0].h / 2;
                    }

                    const childLefts = cBounds.map(cb => ({ x: cb.x, y: cb.y + cb.h / 2 }));
                    const minChildX = Math.min(...childLefts.map(pt => pt.x));
                    const midRatio = group.isAdoptive ? 0.42 : 0.52;
                    const midX = originX + (minChildX - originX) * midRatio;

                    svgLayer.appendChild(createSvgEl('line', {
                        x1: originX, y1: originY, x2: midX, y2: originY,
                        stroke: strokeColor, 'stroke-width': '2', 'stroke-linecap': 'round', ...dashAttrs
                    }));

                    const allY = [originY, ...childLefts.map(pt => pt.y)];
                    svgLayer.appendChild(createSvgEl('line', {
                        x1: midX, y1: Math.min(...allY), x2: midX, y2: Math.max(...allY),
                        stroke: strokeColor, 'stroke-width': '2', 'stroke-linecap': 'round', ...dashAttrs
                    }));

                    childLefts.forEach(pt => {
                        svgLayer.appendChild(createSvgEl('line', {
                            x1: midX, y1: pt.y, x2: pt.x, y2: pt.y,
                            stroke: strokeColor, 'stroke-width': '2', 'stroke-linecap': 'round', ...dashAttrs
                        }));
                    });
                }
            });

            // 3. 人物カードの生成（横書き / 縦書き 対応）
            ids.forEach(id => {
                const p = normalizePersonData(people[id]);
                const b = bounds[id];
                if (!b) return;

                const isSelected = selectedPersonId === id;
                const isDeceased = isPersonDeceased(p);
                const noteClean = (p.note || '').trim();

                const genderAccent = p.gender === 'male'
                    ? 'border-t-4 border-t-sky-500'
                    : p.gender === 'female'
                        ? 'border-t-4 border-t-rose-400'
                        : 'border-t-4 border-t-emerald-400';

                const genderBadge = p.gender === 'male'
                    ? '<span class="w-2 h-2 rounded-full bg-sky-500 shrink-0" title="男性"></span>'
                    : p.gender === 'female'
                        ? '<span class="w-2 h-2 rounded-full bg-rose-400 shrink-0" title="女性"></span>'
                        : '<span class="w-2 h-2 rounded-full bg-emerald-400 shrink-0" title="その他"></span>';

                const birthDisp = formatDisplayDate(p.birthDate);
                const deathDisp = formatDisplayDate(p.deathDate);
                let dateText = '';
                if (birthDisp && deathDisp) {
                    dateText = `${birthDisp} 〜 ${deathDisp}`;
                } else if (birthDisp && isDeceased) {
                    dateText = `${birthDisp} 〜 故人(没年不明)`;
                } else if (birthDisp) {
                    dateText = `${birthDisp} 生`;
                } else if (deathDisp) {
                    dateText = `〜 ${deathDisp} 没`;
                } else if (isDeceased) {
                    dateText = '故人（没年不明）';
                }

                // 分家・相続バッジ
                let statusBadgesHtml = '';
                if (p.branchType && BRANCH_LABELS[p.branchType]) {
                    const bl = BRANCH_LABELS[p.branchType];
                    statusBadgesHtml += `<span class="px-1.5 py-0.5 rounded text-[9px] font-bold ${bl.cls}">${bl.text}</span>`;
                }
                if (p.inheritanceStatus && INHERITANCE_LABELS[p.inheritanceStatus]) {
                    const il = INHERITANCE_LABELS[p.inheritanceStatus];
                    statusBadgesHtml += `<span class="px-1.5 py-0.5 rounded text-[9px] font-bold ${il.cls}">${il.text}</span>`;
                }

                // 関連系統へのジャンプボタン
                let linkedLineageHtml = '';
                if (p.linkedLineageId && appState.lineages[p.linkedLineageId]) {
                    const targetLin = appState.lineages[p.linkedLineageId];
                    linkedLineageHtml = `
                        <button type="button" onclick="event.stopPropagation(); switchLineage('${targetLin.id}')"
                            class="mt-1 inline-flex items-center gap-1 px-2 py-0.5 rounded-md bg-indigo-50 hover:bg-indigo-100 dark:bg-indigo-950/80 dark:hover:bg-indigo-900 text-indigo-700 dark:text-indigo-300 border border-indigo-200 dark:border-indigo-800 text-[10px] font-bold transition-colors cursor-pointer">
                            <span>${escapeHTML(targetLin.name)} 表示</span>
                            <span>→</span>
                        </button>
                    `;
                }

                const displayName = getPersonDisplayName(p);
                const charLen = Array.from(displayName).length;
                const nameFontClass = charLen >= 13
                    ? 'text-[13px]'
                    : charLen >= 10
                        ? 'text-[15px]'
                        : charLen >= 8
                            ? 'text-base'
                            : 'text-[17px]';

                const card = document.createElement('div');
                card.style.left = `${b.x}px`;
                card.style.top = `${b.y}px`;
                card.style.width = `${b.w}px`;
                if (isVertical) {
                    card.style.height = `${b.h}px`;
                } else {
                    card.style.minHeight = `${b.h}px`;
                }

                card.className = `person-card absolute bg-white dark:bg-slate-800 rounded-xl shadow-md border cursor-pointer flex flex-col justify-between p-2.5 text-left ${genderAccent} ${
                    isSelected
                        ? 'border-indigo-500 ring-2 ring-indigo-500/40 shadow-indigo-500/15 z-20'
                        : 'border-slate-200 dark:border-slate-700 hover:border-indigo-300 dark:hover:border-indigo-600 z-10'
                } ${isDeceased ? 'bg-slate-50/95 dark:bg-slate-800/85' : ''}`;

                if (!isVertical) {
                    card.innerHTML = `
                        <div class="w-full text-left">
                            <div class="flex items-center justify-between gap-1.5 mb-0.5">
                                <div class="flex items-center gap-1.5 min-w-0">
                                    ${genderBadge}
                                    <span class="text-[11px] font-bold text-indigo-600 dark:text-indigo-400 truncate">
                                        ${escapeHTML(p.relationship || '親族')}
                                    </span>
                                </div>
                                ${isDeceased ? '<span class="text-[9px] font-bold px-1.5 py-0.2 rounded bg-slate-200 dark:bg-slate-700 text-slate-600 dark:text-slate-300 shrink-0">故人</span>' : ''}
                            </div>
                            <div class="font-bold ${nameFontClass} tracking-tight text-slate-900 dark:text-white leading-snug whitespace-nowrap overflow-hidden text-left">
                                ${escapeHTML(displayName)}
                            </div>
                            ${p.dharmaName ? `<div class="text-[11px] font-serif text-amber-800 dark:text-amber-300/90 mt-0.5 leading-snug text-left">戒名: ${escapeHTML(p.dharmaName)}</div>` : ''}

                            ${p.domicile ? `<div class="text-[10px] text-slate-500 dark:text-slate-400 mt-0.5 leading-snug break-words text-left"><span class="text-slate-400">本籍:</span>${escapeHTML(p.domicile)}</div>` : ''}
                        </div>
                        <div class="mt-1 w-full space-y-1 text-left">
                            ${(statusBadgesHtml || p.branchDetail) ? `
                                <div class="flex flex-wrap items-center justify-start gap-1">
                                    ${statusBadgesHtml}
                                    ${p.branchDetail ? `<span class="text-[10px] text-slate-600 dark:text-slate-300 break-words text-left">${escapeHTML(p.branchDetail)}</span>` : ''}

                        <div class="vertical-card-text w-full h-full flex flex-col justify-start items-start gap-1.5 overflow-hidden">
                            <div class="flex items-center gap-1 shrink-0">
                                ${genderBadge}
                                <span class="text-[10px] font-bold text-indigo-600 dark:text-indigo-400">
                                    ${escapeHTML(p.relationship || '親族')}
                                </span>
                                ${isDeceased ? '<span class="text-[9px] font-bold px-1 py-0.5 rounded bg-slate-200 dark:bg-slate-700 text-slate-600 dark:text-slate-300">故</span>' : ''}
                            </div>
                            <div class="font-bold ${nameFontClass} tracking-wider text-slate-900 dark:text-white leading-snug whitespace-nowrap shrink-0">
                                ${escapeHTML(displayName)}
                            </div>
                            ${p.dharmaName ? `<div class="text-[10px] text-amber-800 dark:text-amber-300/90 leading-snug shrink-0">戒名 ${escapeHTML(p.dharmaName)}</div>` : ''}
                            ${dateText ? `<div class="text-[10px] text-slate-500 dark:text-slate-400 leading-snug shrink-0">${escapeHTML(dateText)}</div>` : ''}
                            ${p.domicile ? `<div class="text-[10px] text-slate-500 dark:text-slate-400 leading-snug break-words">本籍 ${escapeHTML(p.domicile)}</div>` : ''}
                            ${(statusBadgesHtml || p.branchDetail) ? `
                                <div class="flex flex-wrap items-center gap-1 shrink-0">
                                    ${statusBadgesHtml}
                                    ${p.branchDetail ? `<span class="text-[10px] text-slate-600 dark:text-slate-300">${escapeHTML(p.branchDetail)}</span>` : ''}
                                </div>
                            ` : ''}
                            ${noteClean ? `<div class="text-[10px] text-slate-600 dark:text-slate-300 border-r border-slate-200 dark:border-slate-700 pr-1 leading-relaxed break-words">${escapeHTML(noteClean)}</div>` : ''}
                            ${linkedLineageHtml}
                        </div>
                    `;
                }

                card.addEventListener('click', (e) => {
                    e.stopPropagation();
                    openEditModal(id);
                });

                nodesContainer.appendChild(card);
            });
        }

        function createSvgEl(tag, attrs) {
            const el = document.createElementNS('http://www.w3.org/2000/svg', tag);
            for (const [k, v] of Object.entries(attrs)) {
                el.setAttribute(k, v);
            }
            return el;
        }

        function toggleParentAdoptiveStatus(childId, parentId) {
            const people = getActivePeople();
            const child = people[childId];
            if (!child) return;
            normalizePersonData(child);

            if (child.adoptiveParentIds.includes(parentId)) {
                child.adoptiveParentIds = child.adoptiveParentIds.filter(x => x !== parentId);
                showToast('「実親（実線）」に変更しました');
            } else {
                child.adoptiveParentIds.push(parentId);
                showToast('「養親（点線）」に変更しました');
            }
            saveToLocalStorage();
            renderFamilyLinks(childId);
            renderTree();
        }

        function renderFamilyLinks(id) {
            const people = getActivePeople();
            const p = normalizePersonData(people[id]);
            const box = document.getElementById('family-links-box');
            const content = document.getElementById('family-links-content');
            content.innerHTML = '';

            const addLinkRow = (label, relType, targetIds) => {
                const valid = (targetIds || []).filter(tid => people[tid]);
                if (valid.length === 0) return;
                const row = document.createElement('div');
                row.className = 'flex items-center gap-1.5 flex-wrap';
                row.innerHTML = `<span class="text-slate-400 font-semibold">${label}:</span>`;
                valid.forEach(tid => {
                    const chip = document.createElement('div');
                    chip.className = 'inline-flex items-center bg-white dark:bg-slate-700 border border-slate-200 dark:border-slate-600 rounded-md overflow-hidden shadow-2xs';

                    const nameBtn = document.createElement('button');
                    nameBtn.type = 'button';
                    nameBtn.className = 'px-2 py-0.5 text-[11px] font-medium hover:text-indigo-600 dark:hover:text-indigo-300 cursor-pointer';
                    nameBtn.textContent = getPersonDisplayName(people[tid]);
                    nameBtn.onclick = () => openEditModal(tid);
                    chip.appendChild(nameBtn);

                    if (relType === 'parent' || relType === 'child') {
                        const childObj = relType === 'parent' ? p : normalizePersonData(people[tid]);
                        const parentObjId = relType === 'parent' ? tid : id;
                        const isAdopt = isAdoptiveParent(childObj, parentObjId);

                        const kindBtn = document.createElement('button');
                        kindBtn.type = 'button';
                        kindBtn.className = isAdopt
                            ? 'px-1.5 py-0.5 text-[10px] font-bold bg-indigo-50 text-indigo-700 dark:bg-indigo-950 dark:text-indigo-300 border-l border-slate-200 dark:border-slate-600 cursor-pointer'
                            : 'px-1.5 py-0.5 text-[10px] font-bold bg-slate-100 text-slate-600 dark:bg-slate-600 dark:text-slate-200 border-l border-slate-200 dark:border-slate-600 cursor-pointer';
                        kindBtn.title = 'クリックで実親子（実線）と養親子（点線）を切り替え';
                        kindBtn.textContent = isAdopt ? '養(点線)' : '実(実線)';
                        kindBtn.onclick = (e) => {
                            e.stopPropagation();
                            toggleParentAdoptiveStatus(childObj.id, parentObjId);
                        };
                        chip.appendChild(kindBtn);
                    }

                    const unlinkBtn = document.createElement('button');
                    unlinkBtn.type = 'button';
                    unlinkBtn.className = 'px-1.5 py-0.5 text-[10px] text-slate-400 hover:text-rose-600 hover:bg-rose-50 dark:hover:bg-rose-950/60 border-l border-slate-200 dark:border-slate-600 cursor-pointer';

                    unlinkBtn.textContent = '×';

                        e.stopPropagation();
                        unlinkFamilyRelation(id, relType, tid);
                    };

                    chip.appendChild(unlinkBtn);
                    row.appendChild(chip);
                });
                content.appendChild(row);
            };

            addLinkRow('親', 'parent', p.parentIds);
            addLinkRow('配偶者', 'spouse', p.spouseIds);
            addLinkRow('子', 'child', p.childrenIds);

            if (content.children.length === 0) {
                content.innerHTML = '<p class="text-[11px] text-slate-400">まだ家族関係が登録されていません。</p>';
            }

            const targetSelect = document.getElementById('existing-rel-target');
            if (targetSelect) {
                targetSelect.innerHTML = '<option value="">人物を選択...</option>';
                Object.values(people).forEach(other => {
                    if (other.id === id) return;
                    const opt = document.createElement('option');
                    opt.value = other.id;
                    const relLabel = other.relationship ? ` (${other.relationship})` : '';
                    opt.textContent = `${getPersonDisplayName(other)}${relLabel}`;
                    targetSelect.appendChild(opt);
                });
            }

            box.classList.remove('hidden');
        }

        function linkExistingPerson() {
            const currentId = document.getElementById('edit-person-id').value;
            const relType = document.getElementById('existing-rel-type').value;
            const targetId = document.getElementById('existing-rel-target').value;
            const people = getActivePeople();

            if (!currentId || !people[currentId] || !targetId || !people[targetId]) {
                showToast('紐づける人物を選択してください');
                return;
            }
            if (currentId === targetId) return;

            const current = normalizePersonData(people[currentId]);
            const target = normalizePersonData(people[targetId]);

            const pushUnique = (arr, val) => {
                if (!Array.isArray(arr)) return [val];
                if (!arr.includes(val)) arr.push(val);
                return arr;
            };

            if (relType === 'spouse') {
                current.spouseIds = pushUnique(current.spouseIds, targetId);
                target.spouseIds = pushUnique(target.spouseIds, currentId);
                showToast(`${getPersonDisplayName(target)} を配偶者として紐づけました`);
            } else if (relType === 'parent_bio' || relType === 'parent_adopt') {
                current.parentIds = pushUnique(current.parentIds, targetId);
                target.childrenIds = pushUnique(target.childrenIds, currentId);
                if (relType === 'parent_adopt') {
                    current.adoptiveParentIds = pushUnique(current.adoptiveParentIds, targetId);
                    showToast(`${getPersonDisplayName(target)} を養親（点線）として紐づけました`);
                } else {
                    current.adoptiveParentIds = (current.adoptiveParentIds || []).filter(x => x !== targetId);
                    showToast(`${getPersonDisplayName(target)} を実親（実線）として紐づけました`);
                }
            } else if (relType === 'child_bio' || relType === 'child_adopt') {
                current.childrenIds = pushUnique(current.childrenIds, targetId);
                target.parentIds = pushUnique(target.parentIds, currentId);
                if (relType === 'child_adopt') {
                    target.adoptiveParentIds = pushUnique(target.adoptiveParentIds, currentId);
                    showToast(`${getPersonDisplayName(target)} を養子（点線）として紐づけました`);
                } else {
                    target.adoptiveParentIds = (target.adoptiveParentIds || []).filter(x => x !== currentId);
                    showToast(`${getPersonDisplayName(target)} を実子（実線）として紐づけました`);
                }
            }

            saveToLocalStorage();
            renderFamilyLinks(currentId);
            renderTree();
            if (appState.activeBottomTab === 'people') renderPeopleView();
        }

        function unlinkFamilyRelation(currentId, relType, targetId) {
            const people = getActivePeople();
            const current = normalizePersonData(people[currentId]);
            const target = normalizePersonData(people[targetId]);
            if (!current || !target) return;

            if (relType === 'spouse') {
                current.spouseIds = (current.spouseIds || []).filter(x => x !== targetId);
                target.spouseIds = (target.spouseIds || []).filter(x => x !== currentId);
            } else if (relType === 'parent') {
                current.parentIds = (current.parentIds || []).filter(x => x !== targetId);
                current.adoptiveParentIds = (current.adoptiveParentIds || []).filter(x => x !== targetId);
                target.childrenIds = (target.childrenIds || []).filter(x => x !== currentId);
            } else if (relType === 'child') {
                current.childrenIds = (current.childrenIds || []).filter(x => x !== targetId);
                target.parentIds = (target.parentIds || []).filter(x => x !== currentId);
                target.adoptiveParentIds = (target.adoptiveParentIds || []).filter(x => x !== currentId);
            }

            saveToLocalStorage();
            renderFamilyLinks(currentId);
            renderTree();
            if (appState.activeBottomTab === 'people') renderPeopleView();
            showToast(`${getPersonDisplayName(target)} との関係を解除しました`);
        }

        function savePerson() {
            const people = getActivePeople();
            const id = document.getElementById('edit-person-id').value;
            const mode = document.getElementById('edit-relation-mode').value;
            const targetId = document.getElementById('edit-target-id').value;

            const lastName = document.getElementById('field-last-name').value.trim();
            const firstName = document.getElementById('field-first-name').value.trim();
            if (!lastName && !firstName) {
                showToast('名字または名前を入力してください');
                document.getElementById('field-last-name').focus();
                return;
            }

            const deathDateVal = document.getElementById('field-death').value.trim();
            const lifeStatusVal = deathDateVal ? 'deceased' : document.getElementById('field-life-status').value;

            const data = {
                lastName,
                firstName,
                name: getPersonDisplayName({ lastName, firstName }),
                dharmaName: document.getElementById('field-dharma-name').value.trim(),
                relationship: document.getElementById('field-relationship').value.trim(),
                gender: document.getElementById('field-gender').value,
                birthDate: document.getElementById('field-birth').value.trim(),
                deathDate: deathDateVal,
                lifeStatus: lifeStatusVal,
                domicile: document.getElementById('field-domicile').value.trim(),
                linkedLineageId: document.getElementById('field-linked-lineage').value,
                branchType: document.getElementById('field-branch-type').value,
                inheritanceStatus: document.getElementById('field-inheritance').value,
                branchDetail: document.getElementById('field-branch-detail').value.trim(),
                note: document.getElementById('field-note').value.trim(),
                source: document.getElementById('field-source').value.trim()
            };

            if (mode === 'edit' && id && people[id]) {
                Object.assign(people[id], data);
                showToast('人物情報を更新しました');
            } else {
                const newId = 'p_' + Date.now();
                const newPerson = {
                    id: newId,
                    ...data,
                    parentIds: [],
                    adoptiveParentIds: [],
                    spouseIds: [],
                    childrenIds: []
                };

                if (mode === 'spouse' && people[targetId]) {
                    const target = normalizePersonData(people[targetId]);
                    newPerson.spouseIds.push(targetId);
                    target.spouseIds.push(newId);
                    (target.childrenIds || []).forEach(cid => {
                        if (people[cid]) {
                            normalizePersonData(people[cid]);
                            newPerson.childrenIds.push(cid);
                            if (!people[cid].parentIds.includes(newId)) people[cid].parentIds.push(newId);
                            if (isAdoptiveParent(people[cid], targetId) && !people[cid].adoptiveParentIds.includes(newId)) {
                                people[cid].adoptiveParentIds.push(newId);
                            }
                        }
                    });
                } else if (mode === 'parent' && people[targetId]) {
                    const target = normalizePersonData(people[targetId]);
                    const parentKind = document.querySelector('input[name="add-parent-kind"]:checked')?.value || 'biological';
                    const isAdoptive = parentKind === 'adoptive';

                    newPerson.childrenIds.push(targetId);
                    target.parentIds.push(newId);
                    if (isAdoptive) {
                        target.adoptiveParentIds.push(newId);
                    }

                    const sameKindParents = target.parentIds.filter(pid => isAdoptiveParent(target, pid) === isAdoptive);
                    if (sameKindParents.length === 2) {
                        const otherParentId = sameKindParents.find(pid => pid !== newId);
                        if (otherParentId && people[otherParentId]) {
                            if (!newPerson.spouseIds.includes(otherParentId)) newPerson.spouseIds.push(otherParentId);
                            if (!people[otherParentId].spouseIds.includes(newId)) people[otherParentId].spouseIds.push(newId);
                        }
                    }
                } else if (mode === 'child' && people[targetId]) {
                    const target = normalizePersonData(people[targetId]);
                    newPerson.parentIds.push(targetId);
                    target.childrenIds.push(newId);
                    if (target.spouseIds && target.spouseIds.length > 0) {
                        const spouseId = target.spouseIds[0];
                        if (people[spouseId]) {
                            newPerson.parentIds.push(spouseId);
                            people[spouseId].childrenIds.push(newId);
                        }
                    }
                }

                people[newId] = newPerson;
                selectedPersonId = newId;
                showToast(`${data.name} を追加しました`);
            }

            saveToLocalStorage();
            renderLineageTabs();
            closeEditor();
            renderTree();
            if (appState.activeBottomTab === 'people') renderPeopleView();
            if (appState.activeBottomTab === 'map') renderMapView();
        }

        }

        function getPersonYearRange(p) {
            const bParsed = parseDateInput(p.birthDate);
            const dParsed = parseDateInput(p.deathDate);
            const bYear = bParsed && bParsed.year ? bParsed.year : null;
            const dYear = dParsed && dParsed.year ? dParsed.year : null;
            const isDeceased = isPersonDeceased(p);
            const currentYear = new Date().getFullYear();

            if (!bYear && !dYear) return null;

            const startYear = bYear || (dYear - 60);
            let endYear;
            let isOngoing = false;
            let isUnknownDeath = false;

            if (dYear) {
                endYear = dYear;
            } else if (isDeceased) {
                endYear = startYear + 80;
                isUnknownDeath = true;
            } else {
                endYear = currentYear;
                isOngoing = true;
            }

            return {
                startYear,
                endYear: Math.max(startYear, endYear),
                hasExactBirth: !!bYear,
                hasExactDeath: !!dYear,
                isOngoing,
                isUnknownDeath
            };
        }

        function selectDocType(type) {
            currentDocType = type;
            const types = ['photo', 'url', 'pdf'];
            const activeCls = 'py-2 px-2 rounded-xl border text-xs font-bold flex items-center justify-center gap-1.5 transition-all cursor-pointer bg-indigo-50 border-indigo-500 text-indigo-700 dark:bg-indigo-950/70 dark:text-indigo-300';

        function saveDocumentItem() {
            const categoryKey = document.getElementById('doc-category-key').value;
            const title = document.getElementById('doc-title-input').value.trim();
            const memo = document.getElementById('doc-memo-input').value.trim();
            const lineage = getActiveLineage();

            if (!title) {
                showToast('タイトル・名称を入力してください');
                document.getElementById('doc-title-input').focus();
                return;
            }

            if (currentDocType === 'url') {
                const url = document.getElementById('doc-url-input').value.trim();
                if (!url) {
                    showToast('URLを入力してください');
                    document.getElementById('doc-url-input').focus();
                    return;
                }
                lineage.documents[categoryKey].push({
                    id: 'doc_' + Date.now(),
                    type: 'url',
                    title,
                    url,
                    memo,
                    createdAt: new Date().toISOString()
                });
                docsOpenState[categoryKey] = true;
                saveToLocalStorage();
                closeDocAddModal();
                renderDocsView();
                showToast('URL資料を追加しました');
                return;
            }

            const fileInput = currentDocType === 'photo'
                ? document.getElementById('doc-photo-file')
                : document.getElementById('doc-pdf-file');

            const file = fileInput.files && fileInput.files[0];
            if (!file) {
                showToast(currentDocType === 'photo' ? '写真ファイルを選択してください' : 'PDFファイルを選択してください');
                return;
            }

            const reader = new FileReader();
            reader.onload = (e) => {
                lineage.documents[categoryKey].push({
                    id: 'doc_' + Date.now(),
                    type: currentDocType,
                    title,
                    fileName: file.name,
                    dataUrl: e.target.result,
                    memo,
                    createdAt: new Date().toISOString()
                });
                docsOpenState[categoryKey] = true;
                saveToLocalStorage();
                closeDocAddModal();
                renderDocsView();
                showToast(currentDocType === 'photo' ? '写真を追加しました' : 'PDFを追加しました');
            };
            reader.onerror = () => {
                showToast('ファイルの読み込みに失敗しました');
            };
            reader.readAsDataURL(file);
        }

        function openDocumentItem(categoryKey, docId) {
            const lineage = getActiveLineage();
            const list = (lineage.documents && lineage.documents[categoryKey]) || [];
            const item = list.find(d => d.id === docId);
            if (!item) return;

            if (item.type === 'url') {
                const targetUrl = /^https?:\/\//i.test(item.url) ? item.url : 'https://' + item.url;
                window.open(targetUrl, '_blank', 'noopener,noreferrer');
                return;
            }

            const modal = document.getElementById('doc-preview-modal');
            const titleEl = document.getElementById('doc-preview-title');
            const memoEl = document.getElementById('doc-preview-memo');
            const openNewEl = document.getElementById('doc-preview-open-new');
            const bodyEl = document.getElementById('doc-preview-body');

            titleEl.textContent = item.title;
            memoEl.textContent = item.memo || item.fileName || '';
            openNewEl.href = item.dataUrl || '#';

            if (item.type === 'photo') {
                bodyEl.innerHTML = `<img src="${item.dataUrl}" alt="${escapeHTML(item.title)}" class="max-w-full max-h-[72dvh] rounded-xl shadow-md object-contain">`;
            } else if (item.type === 'pdf') {
                bodyEl.innerHTML = `<iframe src="${item.dataUrl}" class="w-full h-[72dvh] rounded-xl border border-slate-200 dark:border-slate-700 bg-white"></iframe>`;
            }

            modal.classList.remove('hidden');
            if (window.lucide) lucide.createIcons();
        }

        function closeDocPreviewModal() {
            document.getElementById('doc-preview-modal').classList.add('hidden');
            document.getElementById('doc-preview-body').innerHTML = '';
        }

        function deleteDocumentItem(categoryKey, docId) {
            const lineage = getActiveLineage();
            const list = (lineage.documents && lineage.documents[categoryKey]) || [];

                }
            );

        let leafletMap = null;
        let savedMarkersLayer = null;
        let draftMarker = null;
        let draftCoords = null; // { lat, lng }

        function initDomicileMapIfNeeded() {
            const mapEl = document.getElementById('domicile-map');
            if (!mapEl || typeof L === 'undefined') return;

            if (!leafletMap) {
                leafletMap = L.map('domicile-map', {
                    center: [35.6895, 139.6917],
                    zoom: 12
                });

                L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
                    attribution: '&copy; OpenStreetMap contributors',
                    maxZoom: 19
                }).addTo(leafletMap);

                savedMarkersLayer = L.layerGroup().addTo(leafletMap);

                leafletMap.on('click', (e) => {
                    setDraftLocation(e.latlng.lat, e.latlng.lng, null, false);
                });
            }

            setTimeout(() => {
                if (leafletMap) leafletMap.invalidateSize();
            }, 80);
        }

        function setDraftLocation(lat, lng, suggestedAddress = null, shouldPan = true) {
            draftCoords = { lat: Number(lat), lng: Number(lng) };
            const coordsLabel = document.getElementById('map-picked-coords-text');
            if (coordsLabel) {
                coordsLabel.textContent = `${draftCoords.lat.toFixed(5)}, ${draftCoords.lng.toFixed(5)}`;
            }

            const addrInput = document.getElementById('map-pin-address-input');
            if (suggestedAddress && addrInput && !addrInput.value.trim()) {
                addrInput.value = suggestedAddress;
            }

            if (leafletMap) {
                if (draftMarker) {
                    draftMarker.setLatLng([draftCoords.lat, draftCoords.lng]);
                } else {
                    draftMarker = L.circleMarker([draftCoords.lat, draftCoords.lng], {
                        radius: 7,
                        color: '#ffffff',
                        weight: 2,
                        fillColor: '#f59e0b',
                        fillOpacity: 0.95
                    }).addTo(leafletMap).bindPopup('登録予定の地点（地図タップで移動可能）');
                }
                if (shouldPan) {
                    leafletMap.setView([draftCoords.lat, draftCoords.lng], Math.max(leafletMap.getZoom(), 15));
                }
            }
        }

        function clearMapDraftPin() {

            if (draftMarker && leafletMap) {

        async function searchMapAddress() {
            const queryInput = document.getElementById('map-address-search-input');
            const resultsBox = document.getElementById('map-search-results');
            const query = (queryInput?.value || '').trim();

            if (!query) {
                showToast('検索する住所・本籍地を入力してください');
                return;
            }

            resultsBox.classList.remove('hidden');
            resultsBox.innerHTML = `<div class="p-2.5 text-slate-500 dark:text-slate-400">住所を検索中...</div>`;

            const candidates = [];

            // 1. 国土地理院 住所検索API（日本国内の住所・地名に強い）
            try {
                const gsiRes = await fetch(`https://msearch.gsi.go.jp/address-search/AddressSearch?q=${encodeURIComponent(query)}`);
                if (gsiRes.ok) {
                    const gsiData = await gsiRes.json();
                    if (Array.isArray(gsiData)) {
                        gsiData.slice(0, 5).forEach(item => {
                            if (item.geometry && Array.isArray(item.geometry.coordinates) && item.geometry.coordinates.length >= 2) {
                                const [lng, lat] = item.geometry.coordinates;
                                const title = item.properties?.title || query;
                                candidates.push({
                                    label: title,
                                    sub: '国土地理院',
                                    lat: Number(lat),
                                    lng: Number(lng)
                                });
                            }
                        });
                    }
                }
            } catch (err) {
                console.warn('GSI geocoding error:', err);
            }

            // 2. 候補が少なければ OpenStreetMap (Nominatim) も補完検索
            if (candidates.length < 3) {
                try {
                    const nomRes = await fetch(`https://nominatim.openstreetmap.org/search?format=json&countrycodes=jp&limit=4&q=${encodeURIComponent(query)}`);
                    if (nomRes.ok) {
                        const nomData = await nomRes.json();
                        if (Array.isArray(nomData)) {
                            nomData.forEach(item => {
                                if (item.lat && item.lon) {
                                    candidates.push({
                                        label: item.display_name,
                                        sub: 'OpenStreetMap',
                                        lat: parseFloat(item.lat),
                                        lng: parseFloat(item.lon)
                                    });
                                }
                            });
                        }
                    }
                } catch (err) {
                    console.warn('Nominatim geocoding error:', err);
                }
            }

            if (candidates.length === 0) {
                resultsBox.innerHTML = `
                    <div class="p-2.5 text-slate-500 dark:text-slate-400 leading-relaxed">
                        一致する候補が見つかりませんでした。旧地名の場合は現代の市区町村名で検索するか、地図を直接タップして位置を指定してください。
                    </div>
                `;
                return;
            }

            // 先頭候補を自動で仮点配置
            setDraftLocation(candidates[0].lat, candidates[0].lng, candidates[0].label, true);

            resultsBox.innerHTML = candidates.map((c, idx) => `
                <button type="button" onclick="selectSearchCandidate(${idx})" class="w-full p-2 text-left hover:bg-emerald-50 dark:hover:bg-emerald-950/50 transition-colors flex items-start justify-between gap-2 cursor-pointer">
                    <div class="min-w-0 flex-1">
                        <div class="font-bold text-slate-800 dark:text-slate-100 truncate">${escapeHTML(c.label)}</div>
                        <div class="text-[10px] text-slate-400">${escapeHTML(c.sub)}（${c.lat.toFixed(4)}, ${c.lng.toFixed(4)}）</div>
                    </div>
                    <span class="text-[10px] font-bold text-emerald-600 dark:text-emerald-400 shrink-0">指定</span>
                </button>
            `).join('');

            window._latestMapSearchCandidates = candidates;
        }

        }

        function saveMapPin() {
            const nameInput = document.getElementById('map-pin-name-input');
            const addrInput = document.getElementById('map-pin-address-input');
            const personSelect = document.getElementById('map-pin-person-select');
            const name = (nameInput?.value || '').trim();
            const address = (addrInput?.value || '').trim();
            const personId = personSelect?.value || '';

            if (!name) {
                showToast('地点の名前を入力してください');
                nameInput.focus();
                return;
            }

            if (!draftCoords) {
                showToast('住所検索または地図タップで地点を選択してください');
                return;

                address,
                lat: draftCoords.lat,
                lng: draftCoords.lng,

            saveToLocalStorage();
            clearMapDraftPin();
            document.getElementById('map-search-results').classList.add('hidden');
            renderMapView();

        function renderMapView() {
            initDomicileMapIfNeeded();

            const lineage = getActiveLineage();
            const people = lineage.people || {};
            const pins = lineage.mapPins || [];

            const headingEl = document.getElementById('map-lineage-heading');
            if (headingEl) headingEl.textContent = `「${lineage.name}」の本籍地マップ`;

            const countBadge = document.getElementById('map-pin-count-badge');
            if (countBadge) countBadge.textContent = `${pins.length}件`;

            // 関連人物プルダウンを更新
            const personSelect = document.getElementById('map-pin-person-select');
            if (personSelect) {
                const prevVal = personSelect.value;
                personSelect.innerHTML = '<option value="">関連人物（任意）</option>';
                Object.values(people).forEach(rawP => {
                    const p = normalizePersonData(rawP);
                    const opt = document.createElement('option');
                    opt.value = p.id;
                    opt.textContent = getPersonDisplayName(p);
                    personSelect.appendChild(opt);
                });
                personSelect.value = prevVal;
            }

            // 登録済み人物の「本籍地」チップを生成
            const chipsBox = document.getElementById('map-domicile-chips');
            if (chipsBox) {
                const domiciles = [];
                Object.values(people).forEach(rawP => {
                    const p = normalizePersonData(rawP);
                    if (p.domicile && p.domicile.trim()) {
                        domiciles.push({
                            personId: p.id,
                            personName: getPersonDisplayName(p),
                            domicile: p.domicile.trim()
                        });
                    }
                });

        // 11. キャンバス操作（パン・ズーム・全体フィット）

        function applyTransform() {
            const stage = document.getElementById('tree-stage');
            stage.style.transform = `translate(${transform.x}px, ${transform.y}px) scale(${transform.scale})`;
            document.getElementById('zoom-level-text').textContent = `${Math.round(transform.scale * 100)}%`;
        }

        function zoomBy(delta, centerX = null, centerY = null) {
            const viewport = document.getElementById('canvas-viewport');
            const rect = viewport.getBoundingClientRect();
            const cx = centerX !== null ? centerX : rect.width / 2;
            const cy = centerY !== null ? centerY : rect.height / 2;

            const oldScale = transform.scale;
            const newScale = Math.min(2.2, Math.max(0.25, oldScale + delta));
            const ratio = newScale / oldScale;

            transform.x = cx - (cx - transform.x) * ratio;
            transform.y = cy - (cy - transform.y) * ratio;
            transform.scale = newScale;
            applyTransform();
        }

        function centerOnPerson(personId) {
            const b = cardBoundsMap[personId];
            if (!b) return;
            const viewport = document.getElementById('canvas-viewport');
            const rect = viewport.getBoundingClientRect();
            transform.x = rect.width / 2 - (b.x + b.w / 2) * transform.scale;
            transform.y = rect.height / 2 - (b.y + b.h / 2) * transform.scale;
            applyTransform();
        }

        function fitToScreen() {
            const boundsList = Object.values(cardBoundsMap);
            if (boundsList.length === 0) {
                transform = { x: 0, y: 0, scale: 1 };
                applyTransform();
                return;
            }

            const viewport = document.getElementById('canvas-viewport');
            const rect = viewport.getBoundingClientRect();
            if (rect.width === 0 || rect.height === 0) return;

            const minX = Math.min(...boundsList.map(b => b.x));
            const minY = Math.min(...boundsList.map(b => b.y));
            const maxX = Math.max(...boundsList.map(b => b.x + b.w));
            const maxY = Math.max(...boundsList.map(b => b.y + b.h));

            const contentW = Math.max(200, maxX - minX);
            const contentH = Math.max(200, maxY - minY);
            const pad = 48;

            const scaleX = (rect.width - pad * 2) / contentW;
            const scaleY = (rect.height - pad * 2) / contentH;
            const scale = Math.min(1.05, Math.max(0.25, Math.min(scaleX, scaleY)));

            transform.scale = scale;
            transform.x = (rect.width - contentW * scale) / 2 - minX * scale;
            transform.y = (rect.height - contentH * scale) / 2 - minY * scale;
            applyTransform();
        }

        function setupCanvasEvents() {
            const viewport = document.getElementById('canvas-viewport');

            viewport.addEventListener('mousedown', (e) => {
                if (e.button !== 0) return;
                isDragging = true;
                dragStart = { x: e.clientX - transform.x, y: e.clientY - transform.y };
            });

            window.addEventListener('mousemove', (e) => {
                if (!isDragging) return;
                transform.x = e.clientX - dragStart.x;
                transform.y = e.clientY - dragStart.y;
                applyTransform();
            });

            window.addEventListener('mouseup', () => {
                isDragging = false;
            });

            viewport.addEventListener('wheel', (e) => {
                e.preventDefault();
                const rect = viewport.getBoundingClientRect();
                const cx = e.clientX - rect.left;
                const cy = e.clientY - rect.top;
                const delta = e.deltaY < 0 ? 0.1 : -0.1;
                zoomBy(delta, cx, cy);
            }, { passive: false });

            viewport.addEventListener('touchstart', (e) => {
                if (e.touches.length === 1) {
                    isDragging = true;
                    dragStart = { x: e.touches[0].clientX - transform.x, y: e.touches[0].clientY - transform.y };
                } else if (e.touches.length === 2) {
                    isDragging = false;
                    const dx = e.touches[0].clientX - e.touches[1].clientX;
                    const dy = e.touches[0].clientY - e.touches[1].clientY;
                    initialPinchDist = Math.hypot(dx, dy);
                    initialPinchScale = transform.scale;
                }
            }, { passive: true });

            viewport.addEventListener('touchmove', (e) => {
                if (e.touches.length === 1 && isDragging) {
                    transform.x = e.touches[0].clientX - dragStart.x;
                    transform.y = e.touches[0].clientY - dragStart.y;
                    applyTransform();
                } else if (e.touches.length === 2 && initialPinchDist) {
                    const dx = e.touches[0].clientX - e.touches[1].clientX;
                    const dy = e.touches[0].clientY - e.touches[1].clientY;
                    const dist = Math.hypot(dx, dy);
                    const newScale = Math.min(2.2, Math.max(0.25, initialPinchScale * (dist / initialPinchDist)));
                    const rect = viewport.getBoundingClientRect();
                    const cx = ((e.touches[0].clientX + e.touches[1].clientX) / 2) - rect.left;
                    const cy = ((e.touches[0].clientY + e.touches[1].clientY) / 2) - rect.top;
                    const ratio = newScale / transform.scale;
                    transform.x = cx - (cx - transform.x) * ratio;
                    transform.y = cy - (cy - transform.y) * ratio;
                    transform.scale = newScale;
                    applyTransform();
                }
            }, { passive: true });

            viewport.addEventListener('touchend', () => {
                isDragging = false;
                initialPinchDist = null;
            });
        }

        function saveToLocalStorage() {
            try {
                localStorage.setItem(STORAGE_KEY, JSON.stringify(appState));
            } catch (e) {
                console.warn('localStorage保存エラー:', e);
            }
        }

        function loadFromLocalStorage() {
            try {
                const raw = localStorage.getItem(STORAGE_KEY);
                if (!raw) return false;
                const parsed = JSON.parse(raw);
                if (parsed && parsed.lineages && Object.keys(parsed.lineages).length > 0) {
                    appState = { ...appState, ...parsed };
                    return true;
                }
            } catch (e) {
                console.warn('localStorage読込エラー:', e);
            }
            return false;
        }

        function exportJSON() {
            const blob = new Blob([JSON.stringify(appState, null, 2)], { type: 'application/json' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `family_tree_${new Date().toISOString().slice(0, 10)}.json`;
            a.click();
            URL.revokeObjectURL(url);
            showToast('全系統のJSONデータを保存しました');
        }

        function importJSON(event) {
            const file = event.target.files && event.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = (e) => {
                try {
                    const parsed = JSON.parse(e.target.result);
                    if (parsed && parsed.lineages && Object.keys(parsed.lineages).length > 0) {
                        appState = { ...appState, ...parsed };
                        saveToLocalStorage();
                        renderLineageTabs();
                        renderTree();
                        if (appState.activeBottomTab === 'people') renderPeopleView();
                        if (appState.activeBottomTab === 'docs') renderDocsView();
                        if (appState.activeBottomTab === 'map') renderMapView();
                        setTimeout(fitToScreen, 60);
                        showToast('JSONデータを読み込みました');
                    } else {
                        showToast('有効な家系図データではありません');
                    }
                } catch (err) {
                    showToast('JSONファイルの解析に失敗しました');
                }
            };
            reader.readAsText(file);
            event.target.value = '';
        }

        async function exportImage() {
            const stage = document.getElementById('tree-stage');
            if (!stage || Object.keys(getActivePeople()).length === 0) {
                showToast('画像出力する人物がいません');
                return;
            }
            showToast('画像を生成しています...');
            const prevTransform = stage.style.transform;
            stage.style.transform = 'translate(0px, 0px) scale(1)';
            try {
                const canvas = await html2canvas(stage, {
                    backgroundColor: document.documentElement.classList.contains('dark') ? '#0f172a' : '#f8fafc',
                    scale: 2
                });
                const link = document.createElement('a');
                link.download = `${getActiveLineage().name}_${new Date().toISOString().slice(0, 10)}.png`;
                link.href = canvas.toDataURL('image/png');
                link.click();
                showToast('画像を保存しました');
            } catch (err) {
                showToast('画像生成に失敗しました');
            } finally {
                stage.style.transform = prevTransform;
            }
        }

        function escapeHTML(str) {
            if (str === null || str === undefined) return '';
            return String(str)
                .replace(/&/g, '&amp;')
                .replace(/</g, '&lt;')
                .replace(/>/g, '&gt;')
                .replace(/"/g, '&quot;')
                .replace(/'/g, '&#39;');
        }
